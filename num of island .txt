#include <stdio.h>
void dfs(char** grid, int m, int n, int r, int c) {
    if (r < 0 || r >= m || c < 0 || c >= n || grid[r][c] == '0') {
        return;
    }
    grid[r][c] = '0';
    dfs(grid, m, n, r - 1, c); 
    dfs(grid, m, n, r + 1, c); 
    dfs(grid, m, n, r, c - 1); 
    dfs(grid, m, n, r, c + 1); 
}
int numIslands(char** grid, int gridSize, int* gridColSize) {
    if (grid == NULL || gridSize == 0) {
        return 0;
    }   
    int m = gridSize;          
    int n = gridColSize[0];    
    int island_count = 0;
    for (int r = 0; r < m; r++) {
        for (int c = 0; c < n; c++) {
            if (grid[r][c] == '1') {
                island_count++;
                dfs(grid, m, n, r, c);
            }
        }
    }
    
    return island_count;
}
