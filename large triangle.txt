#include <stdio.h>
#include <stdlib.h>
#include <math.h>

struct triangle
{
	int a;
	int b;
	int c;
};

typedef struct triangle triangle;

// Function to calculate the squared area of a triangle multiplied by 16 to avoid floats
// 16 * Area^2 = (a+b+c) * (b+c-a) * (a+c-b) * (a+b-c)
double get_sq_area(triangle tr) {
    double p = (tr.a + tr.b + tr.c) / 2.0;
    return p * (p - tr.a) * (p - tr.b) * (p - tr.c);
}

struct triangle
{
	int a;

};

typedef struct triangle triangle;

// Function to calculate the squared area of a triangle multiplied by 16 to avoid floats
// 16 * Area^2 = (a+b+c) * (b+c-a) * (a+c-b) * (a+b-c)
double get_sq_area(triangle tr) {
    double p = (tr.a + tr.b + tr.c) / 2.0;
    return p * (p - tr.a) * (p - tr.b) * (p - tr.c);
}

void sort_by_area(triangle* tr, int n) {
    /**
    * Sort an array of 'n' triangles by their areas from smallest to largest.
    */
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (get_sq_area(tr[j]) > get_sq_area(tr[j + 1])) {
                triangle temp = tr[j];
                tr[j] = tr[j + 1];
                tr[j + 1] = temp;
            }
        }
    }
}

{
	int n;
	scanf("%d", &n);
	triangle *tr = malloc(n * sizeof(triangle));
	for (int i = 0; i < n; i++) {
		scanf("%d%d%d", &tr[i].a, &tr[i].b, &tr[i].c);
	}
	sort_by_area(tr, n);
	for (int i = 0; i < n; i++) {
		printf("%d %d %d\n", tr[i].a, tr[i].b, tr[i].c);
	}
