#include <stdio.h>
#include <stdlib.h>

int main() {
    int n;
    
    // Read the size of the array
    if (scanf("%d", &n) != 1) {
        return 1;
    }
    
    // Dynamically allocate memory for n integers
    int *arr = (int*)malloc(n * sizeof(int));
    if (arr == NULL) {
        return 1; // Return if memory allocation fails
    }
    
    // Read n space-separated integers into the array
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            free(arr);
            return 1;
        }
    }
    
    // Iterate through the array to calculate the sum
    long long sum = 0; // Using long long to prevent integer overflow
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    
    // Print the final sum
    printf("%lld\n", sum);
    
    // Free the dynamically allocated memory
    free(arr);
    
    return 0;
}
